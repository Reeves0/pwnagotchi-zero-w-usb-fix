# ---------------------------------------------------------------------------
# Boot-partition config for USB-gadget on an original Pi Zero W (32-bit)
# Copy these into the files on the FAT "boot" partition of the SD card.
# ---------------------------------------------------------------------------

# ===== config.txt =====
# Add under the [all] section. This is THE key line that makes the PC see it.
dtoverlay=dwc2,dr_mode=peripheral
# (Do NOT rely on plain "dtoverlay=dwc2" — OTG auto-detect may not switch to
#  peripheral/gadget mode, and then the PC sees nothing.)


# ===== cmdline.txt =====
# This file is ONE single line. Append the modules-load token (e.g. after
# "rootwait"). Example (yours will differ in PARTUUID):
#
# console=serial0,115200 console=tty1 root=PARTUUID=xxxxxxxx-02 rootfstype=ext4 fsck.repair=yes rootwait modules-load=dwc2,g_ether


# ===== ssh =====
# Create an EMPTY file literally named "ssh" (no extension) on the boot
# partition to enable the SSH server on first boot.


# ===== userconf.txt (only needed on Bookworm / newer, which has no default user) =====
# One line: username:hashedpassword
# Generate the hash on any machine with:  openssl passwd -6
# Example (user "pi", password "raspberry"):
# pi:$6$...hash...
